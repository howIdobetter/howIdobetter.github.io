Backtrace

这个算比较简单的，基本学习了 CS61B 课程的就能比较理解 Risc-V 汇编语言了，再结合此课程的具体关于栈的知识，基本上就能理解这个实验了。

在 kernel/printf.c 中实现 backtrace 函数就行。

void
backtrace(void)
{
  uint64 fp;
  uint64 stack_top;
  fp = r_fp();
  stack_top = PGROUNDDOWN(fp) + PGSIZE;
  printf("backtrace:\n");
  while (fp < stack_top) {
    uint64 ra = *(uint64 *)(fp-8);
    printf("%p\n", ra);

    fp = *(uint64 *)(fp - 16);
  }
}

Alarm

要完成这个实验，需要先理清用户态、内核态以及 trapframe 之间的关系。用户程序调用 sigalarm 时，会通过 ecall 指令陷入内核。蹦床代码 uservec 会把用户态寄存器保存到当前进程的 trapframe 中，然后跳转到 usertrap，由系统调用处理逻辑执行 sys_sigalarm。sys_sigalarm 的作用只是把闹钟间隔和用户态 handler 的地址记录在 struct proc 中；完成注册之后，这次系统调用就结束了。

之后，当用户程序正常运行并发生定时器中断时，uservec 会再次把被中断时的用户寄存器保存到 p->trapframe 中。usertrap 根据累计的时钟中断次数判断闹钟是否到期。如果到期，内核需要先把当前 trapframe 完整备份到另一个 trapframe 中，再把 p->trapframe->epc 修改为 handler 的用户虚拟地址。随后内核通过 usertrapret、userret 和 sret 返回用户态。由于返回时使用的 epc 已经变成 handler 的地址，因此进程会在用户态开始执行 handler，而不是立即回到原先被中断的位置。

handler 执行过程中会使用并修改用户态寄存器，因此它结束时必须调用 sigreturn 系统调用。sys_sigreturn 将之前保存的 trapframe 恢复到 p->trapframe，从而恢复被定时器中断时的 epc 和全部用户寄存器，使程序可以从原来的位置继续执行。由于系统调用分发代码会把系统调用的返回值写入 p->trapframe->a0，所以 sys_sigreturn 必须返回恢复后的原始 a0，否则原始 a0 会被系统调用返回值覆盖。

usertrap 中的时间中断处理逻辑代码如下：

// give up the CPU if this is a timer interrupt.
  if(which_dev == 2) {
    if (p->alarmstatus == 0 && (p->interval > 0 || p->handler != 0)) {
      p->ticks++;
      if (p->ticks >= p->interval) {
        p->ticks = 0;
        p->alarmstatus = 1;
        memmove(p->alarmtrapframe, p->trapframe, sizeof(struct trapframe));
        p->trapframe->epc = p->handler;
      }
    }
    yield();
  }

两个系统调用函数：

// sigalarm
uint64
sys_sigalarm(void)
{
  int ticks;
  uint64 addr;
  struct proc *p = myproc();

  if(argint(0, &ticks) < 0)
    return -1;

  if(argaddr(1, &addr) < 0)
    return -1;

  p->interval = ticks;
  p->handler = addr;
  p->ticks = 0;
  p->alarmstatus = 0;

  return 0;
}

// return
uint64
sys_sigreturn(void)
{
  struct proc* p = myproc();
  memmove(p->trapframe, p->alarmtrapframe, sizeof(struct trapframe));
  p->alarmstatus = 0;
  return p->trapframe->a0;
}

由于增加了 proc 结构体的字段，同时也是增加了未分配内存的指针，因此我们需要在 proc.c 中的 allocproc 函数中为 alarmtrapframe 分配内存，以及添加 freeproc 函数中释放内存的逻辑：

// Look in the process table for an UNUSED proc.
// If found, initialize state required to run in the kernel,
// and return with p->lock held.
// If there are no free procs, or a memory allocation fails, return 0.
static struct proc*
allocproc(void)
{
  struct proc *p;

  for(p = proc; p < &proc[NPROC]; p++) {
    acquire(&p->lock);
    if(p->state == UNUSED) {
      goto found;
    } else {
      release(&p->lock);
    }
  }
  return 0;

found:
  p->pid = allocpid();

  // Allocate a trapframe page.
  if((p->trapframe = (struct trapframe *)kalloc()) == 0){
    release(&p->lock);
    return 0;
  }

  // Allocate a alarmtrapframe page.
  if ((p->alarmtrapframe = (struct trapframe*)kalloc()) == 0) {
    freeproc(p);
    release(&p->lock);
    return 0;
  }

  // An empty user page table.
  p->pagetable = proc_pagetable(p);
  if(p->pagetable == 0){
    freeproc(p);
    release(&p->lock);
    return 0;
  }

  // Set up new context to start executing at forkret,
  // which returns to user space.
  memset(&p->context, 0, sizeof(p->context));
  p->context.ra = (uint64)forkret;
  p->context.sp = p->kstack + PGSIZE;

  // Set up new time
  p->ticks = 0;

  return p;
}

// free a proc structure and the data hanging from it,
// including user pages.
// p->lock must be held.
static void
freeproc(struct proc *p)
{
  if(p->trapframe)
    kfree((void*)p->trapframe);
  if(p->alarmtrapframe)
    kfree((void*)p->alarmtrapframe);
  p->trapframe = 0;
  if(p->pagetable)
    proc_freepagetable(p->pagetable, p->sz);
  p->pagetable = 0;
  p->sz = 0;
  p->pid = 0;
  p->parent = 0;
  p->name[0] = 0;
  p->chan = 0;
  p->killed = 0;
  p->xstate = 0;
  p->state = UNUSED;
}
