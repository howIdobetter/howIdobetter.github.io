Print a page table

非常简单，打印就行了：

void
printdot(int level)
{
  for (int l = 0; l <= level; l++) {
    printf("..");
    if (l != level) {
      printf(" ");
    }
  }
}

void
helper(pagetable_t pagetable, int level)
{
  if (level > 2) return;
  for (int i = 0; i < 512; i++) {
    pte_t* pte = &pagetable[i];
    if ((*pte & PTE_V)) {
      uint64 pa = PTE2PA(*pte);
      printdot(level);
      printf("%d: pte %p pa %p\n", i, *pte, pa);
      helper((pagetable_t)pa, level + 1);
    }
  }
}

void
vmprint(pagetable_t pagetable)
{
  printf("page table %p\n", pagetable);
  helper(pagetable, 0);
}

A kernel pagetable per process

首先当然是在 proc.h 中加入一个 pagetable 字段：

pagetable_t pagetable; // kernel page table for this process

然后我们需要模仿全局内核页表在 vm.c 中创建一个新的进程内核页表：

/*
 * create a direct-map page table for process p.
*/
pagetable_t
pkvminit()
{
  pagetable_t process_kernel_pagetable = (pagetable_t)kalloc();
  memset(process_kernel_pagetable, 0, PGSIZE);
  // uart registers
  pkvmmap(process_kernel_pagetable, UART0, UART0, PGSIZE, PTE_R | PTE_W);

  // virtio mmio disk interface
  pkvmmap(process_kernel_pagetable, VIRTIO0, VIRTIO0, PGSIZE, PTE_R | PTE_W);

  // CLINT
  pkvmmap(process_kernel_pagetable, CLINT, CLINT, 0x10000, PTE_R | PTE_W);

  // PLIC
  pkvmmap(process_kernel_pagetable, PLIC, PLIC, 0x400000, PTE_R | PTE_W);

  // map kernel text executable and read-only.
  pkvmmap(process_kernel_pagetable, KERNBASE, KERNBASE, (uint64)etext-KERNBASE, PTE_R | PTE_X);

  // map kernel data and the physical RAM we'll make use of.
  pkvmmap(process_kernel_pagetable, (uint64)etext, (uint64)etext, PHYSTOP-(uint64)etext, PTE_R | PTE_W);

  // map the trampoline for trap entry/exit to
  // the highest virtual address in the kernel.
  pkvmmap(process_kernel_pagetable, TRAMPOLINE, (uint64)trampoline, PGSIZE, PTE_R | PTE_X);
  return process_kernel_pagetable;
}

当然，这里的 pkvmmap 函数是一个帮助函数，方便我们映射每个进程的内核页表虚拟地址到物理地址（模仿的是 kvmmap 函数）：

void
pkvmmap(pagetable_t process_kernel_pagetable, uint64 va, uint64 pa, uint64 sz, int perm)
{
  if(mappages(process_kernel_pagetable, va, sz, pa, perm) != 0)
    panic("pkvmmap");
}

此外，我们创建进程的时候需要为其分配内核页表，所以在 allocproc() 中加入：

  // An full kernel page table.
  p->kpagetable = pkvminit();
  if(p->kpagetable == 0){
    freeproc(p);
    release(&p->lock);
    return 0;
  }

  // Allocate a page for the process's own kernel stack.
  // Map it high in memory, followed by an invalid guard
  // page.
  char *pa = kalloc();
  if(pa == 0){
    freeproc(p);
    release(&p->lock);
    return 0;
  }
  uint64 va = KSTACK((int) (p - proc));
  pkvmmap(p->kpagetable, va, (uint64)pa, PGSIZE, PTE_R | PTE_W);
  p->kstack = va;

不过这里需要注意的是，procinit() 会重复分配内核栈，因此将分配内核栈的部分删除。

此外，我们很可能会遇到一个麻烦的报错，就是 virtio_disk_intr status,实际上是因为 kvmpa() 函数一直使用的是全局内核页表，而我们现在每个进程都有自己的内核页表了，所以我们需要修改 kvmpa() 函数，使用当前进程的内核页表：

uint64
kvmpa(uint64 va)
{
  uint64 off = va % PGSIZE;
  pte_t *pte;
  uint64 pa;

  pagetable_t pagetable = kernel_pagetable;

  struct proc *p = myproc();
  if(p != 0 && p->kpagetable != 0)
    pagetable = p->kpagetable;

  pte = walk(pagetable, va, 0);
  if(pte == 0)
    panic("kvmpa");
  if((*pte & PTE_V) == 0)
    panic("kvmpa");

  pa = PTE2PA(*pte);
  return pa + off;
}

最后就是在进程结束之后释放内核页表了，所以在 freeproc() 中加入：

  if(p->kpagetable) {
    proc_free_kernelpagetable(p->kpagetable, p->kstack);
    p->kpagetable = 0;
  }

这里我们需要实现 proc_free_kernelpagetable() 函数，释放内核页表：

// Free a process's kernel page table, and free the
// physical memory if refers to.
void
proc_free_kernelpagetable(pagetable_t kpagetable, uint64 kstack)
{
  uvmunmap(kpagetable, kstack, 1, 1);
  freewalk_noleaf(kpagetable);
}

注意我们尽量不要释放内核页表的叶子节点，因为内核页表的叶子节点是映射到内核的物理内存的，如果有两个进程的内核页表映射到了同一个物理内存，那么释放其中一个进程的内核页表的叶子节点就会导致另一个进程的内核页表失效，所以我们在释放内核页表的时候不释放叶子节点。但是像内核栈这样的内核页表的叶子节点是可以释放的，因为每个进程的内核栈都是独立的，所以我们在释放内核页表的时候释放内核栈的叶子节点。

void
freewalk_noleaf(pagetable_t pagetable)
{
  for(int i = 0; i < 512; i++){
    pte_t pte = pagetable[i];

    if((pte & PTE_V) == 0)
      continue;

    if((pte & (PTE_R | PTE_W | PTE_X)) == 0){
      uint64 child = PTE2PA(pte);
      freewalk_noleaf((pagetable_t)child);
    }
    pagetable[i] = 0;
  }

  kfree((void *)pagetable);
}

Simplify copyin/copyinstr

这个难度有点过于大了，涉及到的方面特别多，并且 bug 也是分容易出现，不是 remapp 就是 kerneltrap，但好歹还是完成了。

首先在 defs.h 增加的两个函数接口为：

int             kuvmcopy_range(pagetable_t pagetable, pagetable_t kpagetable, uint64 oldsz, uint64 newsz);

// vmcopyin.c

然后，发现了上一个试验中的一个小 bug（虽然不影响上面的结果），就是 CLINT 事实上不需要在每个进程的内核页表中映射，因为 CLINT 是全局的，所以在 pkvminit() 中删除了 CLINT 的映射。

最重要的就是 kuvmcopy() 和 kuvmcopy_range() 的实现了，kuvmcopy() 主要是拷贝整个进程的用户页表，而 kuvmcopy_range() 主要是拷贝用户页表的一部分。

int kuvmcopy(pagetable_t pagetable, pagetable_t kpagetable, uint64 sz)
{
  pte_t *pte;
  uint64 pa, i;
  uint flags;
  if (sz > PLIC)
    return -1;
  for (i = 0; i < sz; i += PGSIZE) {
    pte = walk(pagetable, i, 0);
    if(pte == 0)
      panic("kuvmcopy: pte should exist");
    if((*pte & PTE_V) == 0)
      panic("kuvmcopy: page not present");
    pa = PTE2PA(*pte);
    flags = PTE_FLAGS(*pte);
    flags &= ~PTE_U;
    if (mappages(kpagetable, i, PGSIZE, pa, flags) < 0) {
      goto err;
    }
  }
  return 0;

  err:
    uvmunmap(kpagetable, 0, i / PGSIZE, 0);
    return -1;
}

int
kuvmcopy_range(pagetable_t pagetable, pagetable_t kpagetable, uint64 oldsz, uint64 newsz)
{
  pte_t *pte;
  uint64 pa;
  uint64 a;
  uint64 start;
  uint flags;

  if(newsz > PLIC)
    return -1;

  start = PGROUNDUP(oldsz);

  for(a = start; a < newsz; a += PGSIZE){
    pte = walk(pagetable, a, 0);

    if(pte == 0 || (*pte & PTE_V) == 0)
      goto err;

    pa = PTE2PA(*pte);
    flags = PTE_FLAGS(*pte);
    flags &= ~PTE_U;

    if(mappages(kpagetable, a, PGSIZE, pa, flags) != 0)
      goto err;
  }

  return 0;

err:
  if(a > start){
    uvmunmap(kpagetable, start, (a - start) / PGSIZE, 0);
  }

  return -1;
}

// Set up first user process.
void
userinit(void)
{
  struct proc *p;

  p = allocproc();
  initproc = p;

  // allocate one user page and copy init's instructions
  // and data into it.
  uvminit(p->pagetable, initcode, sizeof(initcode));
  p->sz = PGSIZE;

  // prepare for the very first "return" from kernel to user.
  p->trapframe->epc = 0;      // user program counter
  p->trapframe->sp = PGSIZE;  // user stack pointer

  safestrcpy(p->name, "initcode", sizeof(p->name));
  p->cwd = namei("/");

  p->state = RUNNABLE;

  kuvmcopy(p->pagetable, p->kpagetable, p->sz);

  release(&p->lock);
}

以及 sys_sbrk() 涉及到的 growproc() 函数：

int
growproc(int n)
{
  uint64 oldsz, newsz;
  struct proc *p = myproc();

  oldsz = p->sz;
  newsz = oldsz;

  if(n > 0){
    if((uint64)n > PLIC - oldsz)
      return -1;

    newsz = uvmalloc(p->pagetable, oldsz, oldsz + n);
    if(newsz == 0)
      return -1;

    if(kuvmcopy_range(p->pagetable, p->kpagetable, oldsz, newsz) < 0){
      uvmdealloc(p->pagetable, newsz, oldsz);
      return -1;
    }
  } else if(n < 0){
    newsz = oldsz + n;
    uint64 begin = PGROUNDUP(newsz);
    uint64 end = PGROUNDUP(oldsz);

    if(end > begin){
      uvmunmap(p->kpagetable, begin, (end - begin) / PGSIZE, 0);
    }

    newsz = uvmdealloc(p->pagetable, oldsz, newsz);
  }

  p->sz = newsz;
  return 0;
}

之后就是 fork() 函数：

// Create a new process, copying the parent.
// Sets up child kernel stack to return as if from fork() system call.
int
fork(void)
{
  int i, pid;
  struct proc *np;
  struct proc *p = myproc();

  // Allocate process.
  if((np = allocproc()) == 0){
    return -1;
  }

  // Copy user memory from parent to child.
  if(uvmcopy(p->pagetable, np->pagetable, p->sz) < 0){
    freeproc(np);
    release(&np->lock);
    return -1;
  }
  np->sz = p->sz;

  // =========added =========
  // Create direct mapping between user vm and kernel vm
  if (kuvmcopy(np->pagetable, np->kpagetable, np->sz) < 0) {
    freeproc(np);
    release(&np->lock);
    return -1;
  }
  // =========added =========

  np->parent = p;

  // copy saved user registers.
  *(np->trapframe) = *(p->trapframe);

  // Cause fork to return 0 in the child.
  np->trapframe->a0 = 0;

  // increment reference counts on open file descriptors.
  for(i = 0; i < NOFILE; i++)
    if(p->ofile[i])
      np->ofile[i] = filedup(p->ofile[i]);
  np->cwd = idup(p->cwd);

  safestrcpy(np->name, p->name, sizeof(p->name));

  pid = np->pid;

  np->state = RUNNABLE;

  release(&np->lock);

  return pid;
}

最后就是 exec() 函数：

int
exec(char *path, char **argv)
{
  char *s, *last;
  int i, off;
  uint64 argc, sz = 0, sp, ustack[MAXARG+1], stackbase;
  struct elfhdr elf;
  struct inode *ip;
  struct proghdr ph;
  pagetable_t pagetable = 0, oldpagetable;
  struct proc *p = myproc();

  begin_op();

  if((ip = namei(path)) == 0){
    end_op();
    return -1;
  }
  ilock(ip);

  // Check ELF header
  if(readi(ip, 0, (uint64)&elf, 0, sizeof(elf)) != sizeof(elf))
    goto bad;
  if(elf.magic != ELF_MAGIC)
    goto bad;

  if((pagetable = proc_pagetable(p)) == 0)
    goto bad;

  // Load program into memory.
  for(i=0, off=elf.phoff; i<elf.phnum; i++, off+=sizeof(ph)){
    if(readi(ip, 0, (uint64)&ph, off, sizeof(ph)) != sizeof(ph))
      goto bad;
    if(ph.type != ELF_PROG_LOAD)
      continue;
    if(ph.memsz < ph.filesz)
      goto bad;
    if(ph.vaddr + ph.memsz < ph.vaddr)
      goto bad;
    uint64 sz1;
    if((sz1 = uvmalloc(pagetable, sz, ph.vaddr + ph.memsz)) == 0)
      goto bad;
    sz = sz1;
    if(ph.vaddr % PGSIZE != 0)
      goto bad;
    if(loadseg(pagetable, ph.vaddr, ip, ph.off, ph.filesz) < 0)
      goto bad;
  }
  iunlockput(ip);
  end_op();
  ip = 0;

  p = myproc();
  uint64 oldsz = p->sz;

  // Allocate two pages at the next page boundary.
  // Use the second as the user stack.
  sz = PGROUNDUP(sz);
  uint64 sz1;
  if((sz1 = uvmalloc(pagetable, sz, sz + 2*PGSIZE)) == 0)
    goto bad;
  sz = sz1;
  uvmclear(pagetable, sz-2*PGSIZE);
  sp = sz;
  stackbase = sp - PGSIZE;

  // Push argument strings, prepare rest of stack in ustack.
  for(argc = 0; argv[argc]; argc++) {
    if(argc >= MAXARG)
      goto bad;
    sp -= strlen(argv[argc]) + 1;
    sp -= sp % 16; // riscv sp must be 16-byte aligned
    if(sp < stackbase)
      goto bad;
    if(copyout(pagetable, sp, argv[argc], strlen(argv[argc]) + 1) < 0)
      goto bad;
    ustack[argc] = sp;
  }
  ustack[argc] = 0;

  // push the array of argv[] pointers.
  sp -= (argc+1) * sizeof(uint64);
  sp -= sp % 16;
  if(sp < stackbase)
    goto bad;
  if(copyout(pagetable, sp, (char *)ustack, (argc+1)*sizeof(uint64)) < 0)
    goto bad;

  // arguments to user main(argc, argv)
  // argc is returned via the system call return
  // value, which goes in a0.
  p->trapframe->a1 = sp;

  // Save program name for debugging.
  for(last=s=path; *s; s++)
    if(*s == '/')
      last = s+1;
  safestrcpy(p->name, last, sizeof(p->name));

  // Commit to the user image.
  // oldpagetable = p->pagetable;
  // p->pagetable = pagetable;
  // p->sz = sz;
  // p->trapframe->epc = elf.entry;  // initial program counter = main
  // p->trapframe->sp = sp; // initial stack pointer
  // proc_freepagetable(oldpagetable, oldsz);
  // Prepare to commit the new user image.
  oldpagetable = p->pagetable;
  oldsz = p->sz;

  // Remove old user mappings from the process kernel page table.
  // do_free must be 0: the physical pages still belong to oldpagetable.
  if(oldsz > 0){
    uvmunmap(p->kpagetable, 0, PGROUNDUP(oldsz) / PGSIZE, 0);
  }

  // Add mappings for the new user address space.
  if(kuvmcopy(pagetable, p->kpagetable, sz) < 0){
    if(oldsz > 0 && kuvmcopy(oldpagetable, p->kpagetable, oldsz) < 0){
      panic("exec: restore kpagetable");
    }

    goto bad;
  }

  // The new user and kernel mappings are both ready.
  // Commit the new address space.
  p->pagetable = pagetable;
  p->sz = sz;
  p->trapframe->epc = elf.entry;
  p->trapframe->sp = sp;

  // The kernel page table no longer points to these old physical pages,
  // so it is now safe to free them.
  proc_freepagetable(oldpagetable, oldsz);
  if (p->pid == 1)
    vmprint(p->pagetable);
  return argc; // this ends up in a0, the first argument to main(argc, argv)

 bad:
  if(pagetable)
    proc_freepagetable(pagetable, sz);
  if(ip){
    iunlockput(ip);
    end_op();
  }
  return -1;
}
